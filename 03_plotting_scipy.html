def parse_file(filename):
    raw = np.loadtxt(filename, skiprows=2, dtype=str)

from scipy.integrate import quad

def gaussian(x, mu, sigma):
    return np.exp(-(x - mu)**2 / (2 * sigma**2)) / np.sqrt(2 * np.pi * sigma**2)

value, error = quad(gaussian, -np.inf, np.inf, args=(0, 1))

Interpolation: scipy.interpolate.interp1d

  
Builds a function from data points: straight lines ('linear') or cubic polynomials joined smoothly ('cubic').


from scipy.interpolate import interp1d

x_data = np.linspace(0, 10, 8)
y_data = np.sin(x_data)
f_linear = interp1d(x_data, y_data)
f_cubic = interp1d(x_data, y_data, kind='cubic')

x = np.linspace(0, 10, 200)
plt.scatter(x_data, y_data, color='black', label='data', zorder=3)
plt.plot(x, f_linear(x), label='linear')
plt.plot(x, f_cubic(x), label='cubic')
plt.plot(x, np.sin(x), ':', label='sin')
plt.legend()
plt.show()
